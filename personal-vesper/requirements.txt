# Vesper — dependencies
#
# Only the first line is genuinely required. Everything below it is a
# subsystem that switches itself off if you skip it, so install what you
# actually want and ignore the rest.
#
# The installer uses this file. It deliberately does NOT include the camera,
# which needs torch — around 2 GB of wheels for a feature most people never
# switch on. If you want it:  pip install -r requirements-vision.txt

# --- reasoning (required) ---------------------------------------------------
anthropic>=0.40.0
# A current CA bundle. It arrives with anthropic anyway, but it is named
# here because config.py now depends on it directly: some Windows Pythons
# have no usable certificate store, and HTTPS then fails for whichever
# hosts happen to be signed by a root they are missing.
certifi>=2024.2.2

# --- speech in, and the wake word -------------------------------------------
# sounddevice needs PortAudio on the system:
#   Debian/Ubuntu:  sudo apt install portaudio19-dev
#   macOS:          brew install portaudio
#   Windows:        nothing — the wheel bundles it
sounddevice>=0.4.6
numpy>=1.24
faster-whisper>=1.0.0
openwakeword>=0.6.0
onnxruntime>=1.16          # openwakeword's inference backend

# --- speech out -------------------------------------------------------------
# Piper is a binary, not a pip package — see README step 3.
# ElevenLabs needs no package: the executor calls the REST endpoint directly.

# The wake word you choose yourself. openwakeword ships four fixed phrases;
# anything else means training a model in Colab. Porcupine lets you type
# the phrase into a web page and download it -- free for personal,
# non-commercial use. Optional: without a key Vesper falls back to
# "hey Jarvis" rather than refusing to start.
pvporcupine>=3.0
